## Task 1: Abandoned orders stop counting toward the 500-order cap

**Why:** `createOrder` refuses new orders once `p.orders.size >= MAX_ORDERS_PER_SHOW` (500). Orders nobody approves count forever, so anyone can freeze a show by clicking the PayPal button 500 times (30/h per visitor). After this task an order that wasn't authorized within `ORDER_OPEN_MS` (3 h) stops counting and can no longer be authorized; if PayPal did place a hold on it, that hold is queued to be cancelled.

## Task 2: Recover a hold PayPal already made (getAuthorization + server guard)

**Why:** Two money gaps. (a) When PayPal's authorize answer is unreadable, `recoverUnreadable` finds the live hold and **voids** it, even when it is exactly the fan's correct hold; the fan is told to try again and the retry makes a second hold. (b) When our answer to PayPal's authorize was lost and PayPal's request-id memory has lapsed, PayPal answers `422 ORDER_ALREADY_AUTHORIZED`; today that maps to "PayPal didn't place the hold", which also invites a second order. After this task the server reads the full authorization (`GET /v2/payments/authorizations/{id}`), and if it matches this order and the signed price, records it as the fan's pledge exactly like a normal answer. Only a mismatching or unreadable hold is refused and cancelled.

```ts
/**
 * Reads one authorization in full (amount, times, custom_id). Used when a hold exists that we never recorded (an
 * unreadable or lost authorize answer), so the runtime can count it as the fan's pledge instead of cancelling it.
 */
export async function getAuthorization(http: PayPalHttp, authorizationId: string, orderId: string): Promise<Hold> {
  assertPayPalId(authorizationId, "authorization id");
  assertPayPalId(orderId, "order id");
  const a = requireBody(await http.request<AuthBody>({ method: "GET", path: `/v2/payments/authorizations/${authorizationId}` }), "authorization");
  if (a.id !== authorizationId) throw new PayPalError(502, "BAD_RESPONSE_BODY", `asked for authorization ${authorizationId} but PayPal answered with ${a.id === undefined ? "no id" : a.id}`);
  const related = a.supplementary_data?.related_ids?.order_id;
  if (related !== undefined && related !== orderId) throw new PayPalError(502, "BAD_RESPONSE_BODY", `authorization ${authorizationId} belongs to order ${related}, not ${orderId}`);
  try {
    return toHold(orderId, a);
  } catch (e) {
    if (e instanceof PayPalError) throw e;
    throw new PayPalError(502, "BAD_RESPONSE_BODY", e instanceof Error ? e.message : "unreadable authorization");
  }
}
```

```ts
import type { AuthorizeResponse, CreateOrderResponse } from "../../worker/api-types";
import { ApiFailure } from "./api";

export interface PledgeApi {
  createOrder(initials: string): Promise<CreateOrderResponse>;
  authorize(orderId: string): Promise<AuthorizeResponse>;

export type PledgeEvent =
  | { kind: "busy" }
  | { kind: "held"; res: AuthorizeResponse; initials: string; resumed?: boolean }
  | { kind: "cancelled"; message: string }
  | { kind: "failed"; message: string };

/** An order PayPal approved whose answer from our server hasn't arrived yet. */
export interface ApprovedOrder {
  orderId: string;
  initials: string;
}

/** Where that order is kept until the server confirms it, so a reload asks about it instead of starting a new hold. */
export interface OrderMemory {
  get(): ApprovedOrder | null;
  set(o: ApprovedOrder): void;
  clear(): void;
}

export const noMemory: OrderMemory = { get: () => null, set: () => undefined, clear: () => undefined };

/** This tab only (sessionStorage), per show. Storage that is switched off just means no resume after a reload. */
export function sessionMemory(showId: string): OrderMemory {
  const key = `ttr:approved:${showId}`;
  const store = (): Storage | null => {
    try {
      return window.sessionStorage;
    } catch {
      return null;
    }
  };
  return {
    get() {
      try {
        const v = JSON.parse(store()?.getItem(key) ?? "null") as Partial<ApprovedOrder> | null;
        if (!v || typeof v.orderId !== "string" || !/^[A-Za-z0-9-]{1,64}$/.test(v.orderId)) return null;

        return null;
      }
    },
    set(o) {
      try {
        store()?.setItem(key, JSON.stringify(o));
      } catch {
        // Storage off: the in-page retries still run.
      }
    },
    clear() {
      try {
        store()?.removeItem(key);
      } catch {
        // Nothing to clear.
      }

export interface PledgeFlowOptions {
  /** Waits before asking again when the answer to "authorize" was lost on the way (network error or timeout). */
  retryDelaysMs?: number[];
  memory?: OrderMemory;
  /** If PayPal never calls back (for example the popup was blocked), the button unlocks after this long. */
  stallMs?: number;
  sleep?: (ms: number) => Promise<void>;
}

export const CANCELLED = "No hold was placed. You can try again whenever you like.";
export const STALLED = "PayPal didn’t open or didn’t answer. Nothing was charged. Please try again, and allow pop-ups for this site.";
const PAYPAL_TROUBLE = "PayPal didn't answer as expected. Nothing was charged. Please try again.";
const UNKNOWN = "PayPal couldn't finish. Nothing was charged. Please try again.";

/** 1 to 3 letters for the bulb; "YOU" when the fan leaves it blank. The server checks it again. */
export function cleanInitials(v: string): string {
  return (v || "").replace(/[^a-z]/gi, "").slice(0, 3).toUpperCase() || "YOU";
}

export function failureMessage(e: unknown): string {
  if (e instanceof ApiFailure) return e.code === "paypal_error" ? PAYPAL_TROUBLE : e.message;
  return UNKNOWN;
}

const lostOnTheWay = (e: unknown) => e instanceof ApiFailure && e.status === 0;

export class PledgeFlow {
  busy = false;
  done = false;
  /** The initials sent with the order in flight; the confirmation uses these, not whatever the box says later. */
  private sent = "YOU";
  private readonly delays: number[];
  private readonly memory: OrderMemory;
  private readonly stallMs: number;
  private readonly sleep: (ms: number) => Promise<void>;
  private stall: ReturnType<typeof setTimeout> | null = null;

  constructor(
    private readonly api: PledgeApi,
    private readonly emit: (e: PledgeEvent) => void,
    private readonly initials: () => string,
    opts: PledgeFlowOptions = {},
  ) {
    this.delays = opts.retryDelaysMs ?? [1000, 3000];
    this.memory = opts.memory ?? noMemory;
    this.stallMs = opts.stallMs ?? 10 * 60_000;
    this.sleep = opts.sleep ?? ((ms) => new Promise((resolve) => setTimeout(resolve, ms)));
  }

  /** The button checks this before opening PayPal: false while an order is in flight or after the fan is on hold. */
  canStart(): boolean {
    return !this.busy && !this.done;
  }

  /** The PayPal button calls this for every attempt, so a failed or cancelled order is never reused. */
  async createOrder(): Promise<string> {
    if (this.done) throw new Error("This fan is already on hold.");
    if (this.busy) throw new Error("A pledge is already in progress.");
    this.busy = true;
    this.sent = this.initials();
    this.emit({ kind: "busy" });
    try {
      const { orderId } = await this.api.createOrder(this.sent);
      this.armStall();
      return orderId;
    } catch (e) {
      this.fail(e);
      throw e;
    }
  }

  /** PayPal approved this order. A lost answer is asked again on the SAME order; it is never replaced by a new one. */
  async approve(orderId: string): Promise<void> {
    if (this.done) return;
    this.disarmStall();
    this.memory.set({ orderId, initials: this.sent });
    for (let attempt = 0; ; attempt += 1) {
      try {
        this.finish(await this.api.authorize(orderId), false);
        return;
      } catch (e) {
        if (lostOnTheWay(e) && attempt < this.delays.length) {
          await this.sleep(this.delays[attempt] ?? 0);
          continue;
        }
        if (!lostOnTheWay(e)) this.memory.clear();
        this.fail(e);
        return;
      }
    }
  }

  /** On page load: an approved order whose answer never arrived. The server answers "held" if it recorded the hold. */
  async resume(): Promise<void> {
    const o = this.memory.get();
    if (!o || this.done || this.busy) return;
    this.busy = true;
    try {
      const res = await this.api.authorize(o.orderId);
      this.sent = o.initials;
      this.finish(res, true);
    } catch (e) {
      this.busy = false;
      if (!lostOnTheWay(e)) this.memory.clear();
    }
  }

  cancel(): void {
    if (this.done) return;
    this.disarmStall();
    this.busy = false;
    this.emit({ kind: "cancelled", message: CANCELLED });
  }

  /** The SDK's error callback. Ignored when no order of ours is in flight (createOrder already reported its own failure). */
  error(e: unknown): void {
    if (this.done || !this.busy) return;
    this.fail(e);
  }

  /** Test hook path: the same create → authorize calls the button makes, without the popup. */
  async simulate(): Promise<void> {
    try {
      await this.approve(await this.createOrder());
    } catch {
      // Already reported through emit.
    }
  }

  private finish(res: AuthorizeResponse, resumed: boolean): void {
    this.memory.clear();
    this.busy = false;
    this.done = true;
    this.emit(resumed ? { kind: "held", res, initials: this.sent, resumed: true } : { kind: "held", res, initials: this.sent });
  }

  private armStall(): void {
    this.disarmStall();
    this.stall = setTimeout(() => {
      this.stall = null;
      if (this.busy && !this.done) {
        this.busy = false;
        this.emit({ kind: "failed", message: STALLED });
      }
    }, this.stallMs);
  }

  private disarmStall(): void {
    if (this.stall !== null) {
      clearTimeout(this.stall);
      this.stall = null;
    }
  }

  private fail(e: unknown): void {
    this.disarmStall();
    this.busy = false;
    this.emit({ kind: "failed", message: failureMessage(e) });
  }
}
```

```ts
/** Removes every element carrying data-ref-only (with its children). Production builds without test hooks only. */
export function stripRefOnly(html: string): string {
  const open = /<([a-z][a-z0-9]*)\b[^>]*\sdata-ref-only(?=[\s=>/])[^>]*>/i;
  let out = html;
  for (let n = 0; n < 100; n += 1) {
    const m = open.exec(out);
    if (!m) return out;
    const tag = (m[1] ?? "").toLowerCase();
    const tags = new RegExp(`<(/?)${tag}\\b[^>]*>`, "gi");
    tags.lastIndex = m.index + m[0].length;
    let depth = 1;
    let end = -1;
    for (let t = tags.exec(out); t; t = tags.exec(out)) {
      depth += t[1] ? -1 : 1;
      if (depth === 0) {
        end = t.index + t[0].length;
        break;
      }
    }
    if (end < 0) throw new Error(`a data-ref-only <${tag}> has no closing tag`);
    out = out.slice(0, m.index) + out.slice(end);
  }
  throw new Error("more than 100 data-ref-only elements");
}
```

```ts
// Polls pause when nobody has touched the page for a while (free-tier Durable Object requests), and resume on input.
export const IDLE_MS = 5 * 60_000;

export class IdleWatch {
  private last: number;
  constructor(
    private readonly now: () => number = () => Date.now(),
    private readonly limitMs: number = IDLE_MS,
  ) {
    this.last = now();
  }
  touch(): void {
    this.last = this.now();
  }
  idle(): boolean {
    return this.now() - this.last >= this.limitMs;
  }
}

export function watchIdle(limitMs = IDLE_MS): IdleWatch {
  const w = new IdleWatch(() => Date.now(), limitMs);
  const touch = () => w.touch();
  for (const ev of ["pointerdown", "pointermove", "keydown", "wheel", "touchstart", "focus"]) addEventListener(ev, touch, { passive: true });
  document.addEventListener("visibilitychange", () => { if (!document.hidden) touch(); });
  return w;
}
```

````markdown
## How it works

- **The model proposes, the mandate decides.** Every money action (capture, void, payout, refund) passes a plain-code guard that checks it against the frozen, hashed mandate the artist signed. The model writes drafts, explanations and notes; it never moves money, and no money step depends on it being available (templates take over when the daily AI budget is spent).
- **Charging happens at tipping**, inside PayPal's 3-day honour period; holds are valid for 29 days ([PayPal: authorize and capture later](https://developer.paypal.com/docs/checkout/standard/customize/authorization/)). So a campaign window is at most 21 days. Payouts go out at the deadline in one batch.
- **Every money call is idempotent** (`PayPal-Request-Id`, a deterministic payout `sender_batch_id`), and the runtime reconciles with PayPal (GET) before retrying anything whose answer was lost.
````

```markdown
**Project name:** Tip the Room

**Links:** Demo {{SITE_ORIGIN}} · Video {{VIDEO_URL}} · Code https://github.com/ramenprotokol/tip-the-room

## Inspiration
Small shows die on risk: someone fronts the room fee and nobody knows if 15 or 50 people will come. Side Door, which ran about 3,300 shows connecting touring artists with DIY venues, closed at the end of 2025, and in a 2024 survey of 1,700 artists 72% said they made no profit from touring. "Only happens if enough people commit" already works for film screenings. We wanted that for gigs, with an agent that settles the money itself.

## What it does
Fans hold a spot with a PayPal hold, not a charge. The artist signs the money rules once: the fan target, the deadline and the payout split. An agent watches the show: it proposes fixes when pledging is behind (the artist approves), charges every hold the moment the target is reached, and at the deadline pays the room, the opener and the artist in one PayPal payout batch. If the target is missed, it cancels every hold and nobody pays. Every step shows in an agent log that names the signed rule that allowed it, and in a settlement ledger with real PayPal sandbox ids.

## How we built it
- PayPal JS SDK v6 + Orders v2 (`intent: AUTHORIZE`), Payments v2 authorizations (capture, void, get), refunds, Payouts v1 and Webhooks with signature verification, all in the sandbox, through a small idempotent REST client.
- Cloudflare Workers AI (Llama 3.3 70B fp8-fast and Llama 3.1 8B fp8, JSON mode) turns the artist's sentence into a draft, explains pace and proposes options, and writes settlement notes. Code computes every number; a plain-code guard checks every money action against the signed, hashed mandate.
- Cloudflare Workers and SQLite Durable Objects (one per show, alarms for the deadline and pace checks), Cloudflare Pages, AG Grid Community for the ledger, three.js and GSAP for the marquee.
- We evaluated the PayPal Agent Toolkit; its OAuth step doesn't run on Workers, so the agent's tools wrap our REST client with the same guard.

## Challenges
Hold timing (capture inside the 3-day honour period, 29-day validity) shaped the whole design: we charge at tipping and pay out at the deadline. Every money call had to survive lost answers and retries without double-charging or double-paying, so each one is idempotent and reconciled with PayPal before any retry.

## Accomplishments
The model proposes, the mandate decides: the AI can explain and suggest, but it can never move money outside the rules the artist signed, and the show still settles when the AI budget runs out.

## What we learned
Simulated fans can be real sandbox holds (server-side card authorizations), so the demo never fakes a PayPal id.

## What's next
Onboard real artists through PayPal's multiparty platform so money goes straight to each payee, and add opener and venue profiles.

## Built with
```

```ts
// Builds the submission video from the recording: lays the PayPal popup over the main video (picture in picture),
// squeezes the slow waits, plays the popup at 2x, and writes test-results/video/tip-the-room.mp4 (H.264, no audio).
// Fails unless the result is under 3 minutes. Needs ffmpeg and ffprobe on PATH; it never installs anything.
// Run: npm run video:build
import { execFileSync } from "node:child_process";
import { existsSync, readFileSync } from "node:fs";
import type { Timeline } from "../tests/video/captions";

const DIR = "test-results/video";
const SQUEEZED_S = 3;
const FAST = 2;
const LIMIT_S = 175;

const have = (bin: string) => { try { execFileSync("which", [bin], { stdio: "ignore" }); return true; } catch { return false; } };
if (!have("ffmpeg") || !have("ffprobe")) {
  console.error("ffmpeg and ffprobe are needed and were not found. This script never installs anything.");
  process.exit(2);
}
const tl = JSON.parse(readFileSync(`${DIR}/timeline.json`, "utf8")) as Timeline;
const main = `${DIR}/main.webm`;
const popup = `${DIR}/popup.webm`;
const s = (ms: number) => (ms / 1000).toFixed(3);

interface Part { start: number; end: number; speed: number }
const parts: Part[] = [];
let cur = 0;
for (const m of [...tl.segments].sort((a, b) => a.start - b.start)) {
  if (m.start < cur || m.end <= m.start) continue; // overlapping marks: the earlier one wins
  if (m.start > cur) parts.push({ start: cur, end: m.start, speed: 1 });
  const dur = m.end - m.start;
  parts.push({ start: m.start, end: m.end, speed: m.kind === "squeeze" ? Math.max(1, dur / (SQUEEZED_S * 1000)) : FAST });
  cur = m.end;
}
if (cur < tl.total) parts.push({ start: cur, end: tl.total, speed: 1 });
const planned = parts.reduce((n, p) => n + (p.end - p.start) / p.speed, 0) / 1000;
console.log(`${parts.length} parts → about ${planned.toFixed(1)} s`);
if (planned >= LIMIT_S) {
  console.error(`Too long (${planned.toFixed(1)} s). Shorten the dwell() times in tests/video/judge-video.spec.ts and record again.`);
  process.exit(1);
}

const f: string[] = [];
const withPopup = tl.popup !== null && existsSync(popup);
if (withPopup) {
  const p = tl.popup!;
  f.push(`[1:v]setpts=PTS-STARTPTS+${s(p.start)}/TB,scale=560:-2[pp]`);
  f.push(`[0:v][pp]overlay=x=W-w-24:y=24:enable='between(t,${s(p.start)},${s(p.end)})':eof_action=pass[base]`);
} else f.push("[0:v]null[base]");
f.push(`[base]split=${parts.length}${parts.map((_, i) => `[s${i}]`).join("")}`);
parts.forEach((p, i) => f.push(`[s${i}]trim=start=${s(p.start)}:end=${s(p.end)},setpts=(PTS-STARTPTS)/${p.speed.toFixed(4)}[p${i}]`));
f.push(`${parts.map((_, i) => `[p${i}]`).join("")}concat=n=${parts.length}:v=1:a=0,fps=30,format=yuv420p[out]`);

const out = `${DIR}/tip-the-room.mp4`;
execFileSync("ffmpeg", ["-y", "-i", main, ...(withPopup ? ["-i", popup] : []), "-filter_complex", f.join(";"), "-map", "[out]", "-an", "-c:v", "libx264", "-preset", "medium", "-crf", "20", "-movflags", "+faststart", out], { stdio: "inherit" });
const dur = Number(execFileSync("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "default=nw=1:nk=1", out]).toString().trim());
console.log(`${out}: ${dur.toFixed(1)} s, no audio track`);
if (!(dur > 0 && dur < 180)) {
  console.error("The finished video must be under 3 minutes.");
  process.exit(1);
}
```
